function displayMessages(data) { const source = Array.isArray(data) ? data.map((item, i) => [item?.id || String(i), item]) : Object.entries(data || {}); const entries = source .map(([id, item]) => { if (typeof item === "string") { item = { sender: "BOT", message: item }; } if (!item || typeof item !== "object") return null; let message = item.message ?? item.text ?? item.content ?? item.body ?? item.msg ?? ""; if (typeof message !== "string") { try { message = JSON.stringify(message ?? ""); } catch { message = String(message); } } let timestamp = item.timestamp ?? ""; if (typeof timestamp === "number" || /^\d+$/.test(String(timestamp))) { const n = Number(timestamp); timestamp = new Date( n < 1e12 ? n * 1000 : n ).toISOString(); } return { id, sender: item.sender || item.from || "Unknown sender", message, timestamp: String(timestamp || "") }; }) .filter(item => item && item.message.trim() !== "") .sort((a, b) => a.timestamp.localeCompare(b.timestamp) ) .slice(-50); $("messageCount").textContent = entries.length + " recent messages"; const container = $("messages"); container.replaceChildren(); if (!entries.length) { container.textContent = "No messages yet."; return; } for (const item of entries) { const block = document.createElement("div"); block.className = "message"; const sender = document.createElement("div"); sender.className = "sender"; sender.textContent = item.sender; const time = document.createElement("div"); time.className = "time"; const date = item.timestamp ? new Date(item.timestamp) : null; time.textContent = date && !Number.isNaN(date.getTime()) ? date.toLocaleString() : "Time unavailable"; const body = document.createElement("div"); body.textContent = item.message; block.append(sender, time, body); container.append(block); } }
